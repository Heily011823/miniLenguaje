# Programa 11: Funciones que llaman funciones, uso antes de declarar y recursion
anotar(cuadrado_suma(2, 3));

trayecto suma(a, b):
  regresar a + b;
fin_sendero

trayecto cuadrado_suma(a, b):
  regresar cuadrado(suma(a, b)) + suma(1, 1);
fin_sendero

trayecto cuadrado(x):
  regresar x * x;
fin_sendero

trayecto fib(n):
  evaluar n < 2:
    regresar n;
  fin_sendero
  regresar fib(n - 1) + fib(n - 2);
fin_sendero

trayecto cuenta_regresiva(n):
  evaluar n == 0:
    regresar "listo";
  fin_sendero
  regresar cuenta_regresiva(n - 1);
fin_sendero

trayecto sin_retorno():
  cargar x en 1;
fin_sendero

anotar("fib(15) = " + fib(15));
anotar(cuenta_regresiva(3000));
anotar(sin_retorno());
anotar(suma(suma(1, 2), suma(3, 4)) * 2);
