# Programa 14: bifurcar (switch) y rondar (do-while)
cargar dia en 3;
bifurcar dia:
  mirador 1:
    anotar("lunes");
  mirador 3:
    anotar("miercoles");
  mirador 5:
    anotar("viernes");
  sendero_base:
    anotar("otro dia");
fin_sendero

bifurcar "b":
  mirador "a":
    anotar("es a");
  mirador "b":
    anotar("es b");
    detener;
    anotar("no se imprime");
  sendero_base:
    anotar("ni a ni b");
fin_sendero

bifurcar 99:
  mirador 1:
    anotar("uno");
fin_sendero
anotar("sin coincidencia ok");

cargar n en 0;
ascender:
  n en n + 1;
  anotar("n = " + n);
repetir_si n < 3;

cargar z en 10;
ascender:
  anotar("se ejecuta una vez aunque sea falso");
repetir_si z < 5;

cargar t en 0;
ascender:
  t en t + 1;
  evaluar t == 2:
    avanzar;
  fin_sendero
  evaluar t == 4:
    detener;
  fin_sendero
  anotar("t = " + t);
repetir_si t < 10;
