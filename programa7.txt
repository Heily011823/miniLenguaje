# Programa 7: Errores de ejecucion capturados y recursion
cargar kilometros en [10, 20, 30];
cargar total en 0;

explorar:
    anotar(100 / total);
imprevisto e:
    anotar("Capturado: " + e);
recoger:
    anotar("Primera verificacion terminada");
fin_sendero

explorar:
    anotar(kilometros[5]);
imprevisto e:
    anotar("Capturado: " + e);
fin_sendero

trayecto factorial(n):
    evaluar n <= 1:
        regresar 1;
    fin_sendero
    regresar n * factorial(n - 1);
fin_sendero

anotar("Factorial de 5: " + factorial(5));
