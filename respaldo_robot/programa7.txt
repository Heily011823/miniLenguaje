# Programa 7: Errores de ejecucion capturados y recursion
cargar paquetes en [10, 20, 30];
cargar total en 0;

intentar_ruta:
    notificar(100 / total);
obstaculo e:
    notificar("Capturado: " + e);
destino:
    notificar("Primera verificacion terminada");
fin_ruta

intentar_ruta:
    notificar(paquetes[5]);
obstaculo e:
    notificar("Capturado: " + e);
fin_ruta

tarea factorial(n):
    evaluar n <= 1:
        entregar 1;
    fin_ruta
    entregar n * factorial(n - 1);
fin_ruta

notificar("Factorial de 5: " + factorial(5));
