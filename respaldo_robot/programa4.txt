# Programa 4: Patrulla con lectura, desvios, detener y avanzar
cargar contador en 0;
cargar limite en 0;

notificar("Cuantas paradas tiene la ruta?");
escanear(limite);

patrullar contador < limite:
    contador en contador + 1;

    evaluar contador % 2 == 0:
        notificar("Parada " + contador + ": pasillo par, se omite");
        avanzar;
    desviar contador == 5:
        notificar("Parada 5: obstaculo en la ruta, el robot se detiene");
        detener;
    alternativa:
        notificar("Entregando en la parada " + contador);
    fin_ruta
fin_ruta

notificar("Recorrido terminado en la parada " + contador);
