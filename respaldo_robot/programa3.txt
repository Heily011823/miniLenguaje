# Programa 3: Modulo de Bateria y Excepciones
modulo RobotEntrega:
    ensamblar(id_robot, bateria_actual):
        mismo.id_robot en id_robot;
        mismo.bateria_actual en bateria_actual;
    fin_ruta

    tarea consumir_bateria(costo):
        evaluar costo > mismo.bateria_actual:
            alerta "Bateria insuficiente para completar el trayecto";
        fin_ruta
        mismo.bateria_actual en mismo.bateria_actual - costo;
        entregar mismo.bateria_actual;
    fin_ruta
fin_ruta

cargar bot en desplegar RobotEntrega("RutaBot-01", 100);

intentar_ruta:
    notificar("Bateria restante: " + bot.consumir_bateria(30));
    notificar("Bateria restante: " + bot.consumir_bateria(200));
obstaculo err:
    notificar("Alerta de Ruta: " + err);
destino:
    notificar("Monitoreo de telemetria finalizado");
fin_ruta
