# Programa 11: Funciones que llaman funciones, uso antes de declarar y recursion
notificar(cuadrado_suma(2, 3));

tarea suma(a, b):
  entregar a + b;
fin_ruta

tarea cuadrado_suma(a, b):
  entregar cuadrado(suma(a, b)) + suma(1, 1);
fin_ruta

tarea cuadrado(x):
  entregar x * x;
fin_ruta

tarea fib(n):
  evaluar n < 2:
    entregar n;
  fin_ruta
  entregar fib(n - 1) + fib(n - 2);
fin_ruta

tarea cuenta_regresiva(n):
  evaluar n == 0:
    entregar "listo";
  fin_ruta
  entregar cuenta_regresiva(n - 1);
fin_ruta

tarea sin_retorno():
  cargar x en 1;
fin_ruta

notificar("fib(15) = " + fib(15));
notificar(cuenta_regresiva(3000));
notificar(sin_retorno());
notificar(suma(suma(1, 2), suma(3, 4)) * 2);
