# suma de pares con avanzar y detener (se corta al llegar a un valor mayor que 100)
cargar valores en [2, 3, 4, 5, 6, 200, 8];
cargar suma en 0;
recorrer i partiendo_de 0 llegando_a longitud(valores) - 1:
  evaluar valores[i] % 2 == 1:
    avanzar;
  fin_ruta
  evaluar valores[i] > 100:
    detener;
  fin_ruta
  suma en suma + valores[i];
fin_ruta
notificar("Suma de pares antes del tope: " + suma);
