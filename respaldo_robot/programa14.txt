# Programa 14: bifurcar (switch) y rondar (do-while)
cargar dia en 3;
bifurcar dia:
  punto_ruta 1:
    notificar("lunes");
  punto_ruta 3:
    notificar("miercoles");
  punto_ruta 5:
    notificar("viernes");
  ruta_base:
    notificar("otro dia");
fin_ruta

bifurcar "b":
  punto_ruta "a":
    notificar("es a");
  punto_ruta "b":
    notificar("es b");
    detener;
    notificar("no se imprime");
  ruta_base:
    notificar("ni a ni b");
fin_ruta

bifurcar 99:
  punto_ruta 1:
    notificar("uno");
fin_ruta
notificar("sin coincidencia ok");

cargar n en 0;
rondar:
  n en n + 1;
  notificar("n = " + n);
repetir_si n < 3;

cargar z en 10;
rondar:
  notificar("se ejecuta una vez aunque sea falso");
repetir_si z < 5;

cargar t en 0;
rondar:
  t en t + 1;
  evaluar t == 2:
    avanzar;
  fin_ruta
  evaluar t == 4:
    detener;
  fin_ruta
  notificar("t = " + t);
repetir_si t < 10;
